    12	function assertRemoteAvailable(
    13	  offlineMessage: string,
    14	  tripId?: string,
    15	  action: OfflineErrorOptions['action'] = 'ACTIVATE_PROMPT',
    16	) {
    17	  requireRemoteSession();
    18	  const status = networkStore.realStatus;
    19	
    20	  if (status === 'online') {
    21	    return;
    22	  }
    23	
    24	  throw new OfflineError(
    25	    status === 'unknown' ? '인터넷 연결을 아직 확인하지 못했어요. 잠시 후 다시 시도해주세요.' : offlineMessage,
    26	    { action: status === 'unknown' ? 'ONLINE_REQUIRED' : action, tripId },
    27	  );
    28	}

    39	export async function routeTripQuery<T>(operations: RouteOperations<T>): Promise<T> {
    40	  requireLocalUserId();
    41	  if (networkStore.realStatus === 'online' && useAuthStore.getState().status === 'signed-in') {
    42	    const sessionId = useAuthStore.getState().sessionId;
    43	    try {
    44	      return await operations.remote();
    45	    } catch (error) {
    46	      // 연결 감지는 API 서버의 응답까지 보장하지 않는다. 원격 접근 실패에만 로컬 이용을 유지한다.
    47	      const unavailable =
    48	        error instanceof APIError &&
    49	        ((error.status === 0 && error.code === 'NETWORK_ERROR') || error.status === 408 || error.status >= 500);
    50	      if (!unavailable || !(await hasAnyActivatedTrip()) || !useAuthStore.isCurrentSession(sessionId)) throw error;
    51	      return await operations.local();
    52	    }
    53	  }
    54	  if (await hasAnyActivatedTrip()) return await operations.local();
    55	  assertRemoteAvailable('오프라인에서는 활성화된 여행만 볼 수 있어요');
    56	  return await operations.remote();
    57	}

    61	  requireLocalUserId();

    63	
    64	  if (isActivated) {

    66	  }
    67	
    68	  assertRemoteAvailable('오프라인에서는 활성화된 여행만 볼 수 있어요', tripId);
    69	  return await operations.remote();
    70	}

      } finally {
        setIsSyncing(false);

    },

  /**
   * 최초 online과 이후 online 전환마다 한 번 동기화
   */
  useEffect(() => {
    const previousStatus = previousStatusRef.current;
    previousStatusRef.current = networkStatus;

    // 최초 unknown 확정과 오프라인 복구 모두 실제 online에서 시작한다.
    if (previousStatus !== 'online' && networkStatus === 'online') {
      if (previousStatus === null) {
        console.log('🚀 [SyncProvider] Initial sync on app start');
        executeSync('App startup');
      } else {
        console.log('🌐 [SyncProvider] Network recovered → Starting sync');
        executeSync('Network recovery');
      }
    }

    // 온라인 → 오프라인 전환 로그
    if (previousStatus === 'online' && networkStatus === 'offline') {
      console.log('📴 [SyncProvider] Network offline → Sync paused');
    }
  }, [networkStatus, executeSync]);

  return null;

/** Trip 조회의 기존 분기는 전역 활성 여행 존재 여부를 사용한다. */
export async function routeTripQuery<T>(operations: RouteOperations<T>): Promise<T> {
  const hasActivated = await hasAnyActivatedTrip();

  if (hasActivated) {
    return await operations.local();
  }

  assertRemoteAvailable('오프라인에서는 활성화된 여행만 볼 수 있어요');
  return await operations.remote();
}

/** Schedule·Expense 조회는 소속 여행의 활성 여부로 분기한다. */
export async function routeChildQuery<T>(tripId: string, operations: RouteOperations<T>): Promise<T> {
  const isActivated = await getTripActivationStatus(tripId);

  if (isActivated) {
    return await operations.local();
  }

  assertRemoteAvailable('오프라인에서는 활성화된 여행만 볼 수 있어요', tripId);
  return await operations.remote();
}
